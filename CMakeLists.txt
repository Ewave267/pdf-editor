cmake_minimum_required(VERSION 3.20)

project(PdfFormEditor VERSION 0.1.0 LANGUAGES CXX)

option(PDF_EDITOR_BUILD_VIEWER "Build the Linux Qt/QML PDF viewer" OFF)
if(PDF_EDITOR_BUILD_VIEWER)
    find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
    set(CMAKE_AUTOMOC ON)
    set(CMAKE_AUTORCC ON)
    add_executable(pdf-form-editor src/app/viewer_main.cpp qml/viewer.qrc)
else()
    add_executable(pdf-form-editor src/app/main.cpp)
endif()
target_compile_features(pdf-form-editor PRIVATE cxx_std_20)
set_target_properties(pdf-form-editor PROPERTIES CXX_EXTENSIONS OFF)

if(MSVC)
    target_compile_options(pdf-form-editor PRIVATE /W4)
else()
    target_compile_options(pdf-form-editor PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)

option(PDF_EDITOR_BUILD_XFA_PROBE "Build the isolated Linux PDFium XFA integration probe" OFF)
if(PDF_EDITOR_BUILD_XFA_PROBE OR PDF_EDITOR_BUILD_VIEWER)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "The PDF worker currently requires the Linux Bubblewrap sandbox")
    endif()
    add_library(pdf-worker-policy STATIC src/pdf/WorkerPolicy.cpp src/pdf/WorkerPolicy.h)
    target_compile_features(pdf-worker-policy PUBLIC cxx_std_20)
    target_include_directories(pdf-worker-policy PUBLIC src)
    target_compile_options(pdf-worker-policy PRIVATE -Wall -Wextra -Wpedantic)
    find_package(PDFium CONFIG REQUIRED)
    if(NOT PDFium_VERSION VERSION_EQUAL "157.0.8086.0")
        message(FATAL_ERROR "The probe requires pinned PDFium 157.0.8086.0")
    endif()
    if(NOT PDF_EDITOR_PDFIUM_PATCHSET STREQUAL "1")
        message(FATAL_ERROR "The probe requires the persistence fix: run python3 tools/build_pdfium.py and use .deps/pdfium-patched")
    endif()
    file(READ "${PDFium_DIR}/args.gn" pdfium_args)
    if(NOT pdfium_args MATCHES "pdf_enable_xfa = true" OR
       NOT pdfium_args MATCHES "pdf_enable_v8 = true")
        message(FATAL_ERROR "PDFium must be built with both XFA and V8 enabled")
    endif()
endif()

if(PDF_EDITOR_BUILD_XFA_PROBE)
    find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
    find_program(NODE_EXECUTABLE node REQUIRED)
    execute_process(COMMAND "${NODE_EXECUTABLE}" --version OUTPUT_VARIABLE node_version
        OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE node_status)
    string(REGEX REPLACE "^v" "" node_version "${node_version}")
    if(NOT node_status EQUAL 0 OR node_version VERSION_LESS "24.0.0")
        message(FATAL_ERROR "The pinned independent reader requires Node.js 24 or newer")
    endif()
    set(PDF_EDITOR_PDFJS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/tests/reader/node_modules/pdfjs-dist"
        CACHE PATH "Pinned PDF.js package for independent XFA validation")
    if(NOT EXISTS "${PDF_EDITOR_PDFJS_DIR}/legacy/build/pdf.mjs")
        message(FATAL_ERROR "Independent reader is missing: run npm ci --prefix tests/reader --ignore-scripts")
    endif()
    find_program(BWRAP_EXECUTABLE bwrap REQUIRED)
    add_library(pdf-xfa-probe STATIC src/pdf/XfaProbe.cpp)
    target_compile_features(pdf-xfa-probe PUBLIC cxx_std_20)
    target_include_directories(pdf-xfa-probe PUBLIC src)
    target_compile_definitions(pdf-xfa-probe PRIVATE PDF_ENABLE_XFA PDF_ENABLE_V8)
    target_link_libraries(pdf-xfa-probe PRIVATE pdfium)
    target_compile_options(pdf-xfa-probe PRIVATE -Wall -Wextra -Wpedantic)
    set_target_properties(pdf-xfa-probe PROPERTIES CXX_EXTENSIONS OFF)
    add_executable(xfa-probe-worker src/app/xfa_probe_main.cpp)
    target_link_libraries(xfa-probe-worker PRIVATE pdf-xfa-probe pdf-worker-policy)
endif()

if(PDF_EDITOR_BUILD_VIEWER)
    find_program(BWRAP_EXECUTABLE bwrap REQUIRED)
    add_library(pdf-viewer STATIC src/pdf/PdfSave.cpp src/pdf/PdfDocument.cpp src/pdf/PdfDocument.h
        src/ui/PdfPageItem.cpp src/ui/PdfPageItem.h
        src/content/AddedContent.cpp src/content/AddedContent.h
        src/ui/AddedOverlay.cpp src/ui/AddedOverlay.h
        src/ui/FormInput.cpp src/ui/FormInput.h)
    target_compile_features(pdf-viewer PUBLIC cxx_std_20)
    target_include_directories(pdf-viewer PUBLIC src)
    target_link_libraries(pdf-viewer PUBLIC Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick)
    get_target_property(pdfium_library pdfium IMPORTED_LOCATION)
    target_compile_definitions(pdf-viewer PRIVATE PDFIUM_LIBRARY_PATH="${pdfium_library}")
    target_compile_options(pdf-viewer PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(pdf-form-editor PRIVATE pdf-viewer Qt6::QuickControls2)
    add_executable(pdf-render-worker src/pdf/PdfRenderWorker.cpp)
    target_compile_features(pdf-render-worker PRIVATE cxx_std_20)
    target_compile_definitions(pdf-render-worker PRIVATE PDF_ENABLE_XFA PDF_ENABLE_V8)
    target_link_libraries(pdf-render-worker PRIVATE pdfium Qt6::Core Qt6::Gui pdf-worker-policy)
    target_compile_options(pdf-render-worker PRIVATE -Wall -Wextra -Wpedantic)
    add_dependencies(pdf-form-editor pdf-render-worker)
endif()

if(BUILD_TESTING)
    add_test(NAME foundation-smoke COMMAND pdf-form-editor --version)
    set_tests_properties(foundation-smoke PROPERTIES
        PASS_REGULAR_EXPRESSION "PDF Form Editor 0.1.0")
    if(PDF_EDITOR_BUILD_VIEWER)
        find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
        add_executable(viewer-tests tests/test_viewer.cpp qml/viewer.qrc)
        target_link_libraries(viewer-tests PRIVATE pdf-viewer Qt6::Test Qt6::QuickControls2)
        target_compile_definitions(viewer-tests PRIVATE TEST_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
        if(PDF_EDITOR_BUILD_XFA_PROBE)
            target_compile_definitions(viewer-tests PRIVATE
                SAVE_NODE="${NODE_EXECUTABLE}" SAVE_PDFJS="${PDF_EDITOR_PDFJS_DIR}"
                SAVE_PDFIUM="${PDFium_DIR}" SAVE_PYTHON="${Python3_EXECUTABLE}")
            add_dependencies(viewer-tests xfa-probe-worker)
        endif()
        find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
        find_package(Threads REQUIRED)
        add_executable(viewer-safety-tests tests/test_viewer_safety.cpp)
        target_link_libraries(viewer-safety-tests PRIVATE pdf-viewer Qt6::Test)
        target_compile_definitions(viewer-safety-tests PRIVATE TEST_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
        target_compile_options(viewer-safety-tests PRIVATE -Wall -Wextra -Wpedantic)
        add_dependencies(viewer-safety-tests pdf-render-worker)
        add_test(NAME viewer-safety COMMAND viewer-safety-tests)
        set_tests_properties(viewer-safety PROPERTIES TIMEOUT 180
            ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" LABELS "viewer;regression;security")
        add_executable(sandbox-policy-worker tests/sandbox_policy_worker.cpp)
        target_link_libraries(sandbox-policy-worker PRIVATE pdf-worker-policy Threads::Threads)
        target_compile_options(sandbox-policy-worker PRIVATE -Wall -Wextra -Wpedantic)
        add_test(NAME worker-safety
            COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_worker_safety.py"
                "$<TARGET_FILE:pdf-render-worker>" "$<TARGET_FILE:sandbox-policy-worker>" "${PDFium_DIR}")
        set_tests_properties(worker-safety PROPERTIES TIMEOUT 60 LABELS "regression;security")
        add_dependencies(viewer-tests pdf-render-worker)
        add_test(NAME viewer-integration COMMAND viewer-tests)
        set_tests_properties(viewer-integration PROPERTIES TIMEOUT 90
            ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" LABELS "viewer;milestone")
    endif()
    if(PDF_EDITOR_BUILD_XFA_PROBE)
        add_test(NAME xfa-regressions
            COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_xfa_probe.py"
                "$<TARGET_FILE:xfa-probe-worker>" "${PDFium_DIR}")
        set_tests_properties(xfa-regressions PROPERTIES TIMEOUT 90 LABELS "regression;security")
        add_test(NAME xfa-roundtrip
            COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_xfa_roundtrip.py"
                "$<TARGET_FILE:xfa-probe-worker>" "${PDFium_DIR}" "${NODE_EXECUTABLE}" "${PDF_EDITOR_PDFJS_DIR}")
        set_tests_properties(xfa-roundtrip PROPERTIES TIMEOUT 90 LABELS "milestone;independent-reader")
    endif()
endif()
